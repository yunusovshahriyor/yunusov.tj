## Код дар видео (санҷидашуда дар Node.js)
```js
let tasks = [];

function addTask(text) {
  tasks.push({ text: text, done: false });
}

function completeTask(i) {
  tasks[i].done = true;
}

function showTasks() {
  for (let i = 0; i < tasks.length; i++) {
    let mark = "☐";
    if (tasks[i].done) { mark = "☑"; }
    console.log(mark + " " + tasks[i].text);
  }
}

addTask("нон харидан");
addTask("дарс хондан");
completeTask(0);
showTasks();
// ☑ нон харидан
// ☐ дарс хондан
```
