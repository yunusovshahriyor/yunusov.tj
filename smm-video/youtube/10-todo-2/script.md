## Код (санҷидашуда дар Node.js)
```js
let tasks = [{ text: "нон харидан", done: false }, { text: "дарс хондан", done: false }];

function showTasks() {
  for (let i = 0; i < tasks.length; i++) {
    if (tasks[i].done) {
      console.log("☑ " + tasks[i].text);
    } else {
      console.log("☐ " + tasks[i].text);
    }
  }
}

function addTask(text) {
  tasks.push({ text: text, done: false });
}

function completeTask(index) {
  if (index >= tasks.length) {
    console.log("Чунин вазифа нест");
    return;
  }
  tasks[index].done = true;
}

function countDone() {
  let count = 0;
  for (let i = 0; i < tasks.length; i++) {
    if (tasks[i].done) {
      count = count + 1;
    }
  }
  return count;
}

showTasks();            // ☐ нон харидан  ☐ дарс хондан
addTask("занг задан");
completeTask(0);
showTasks();            // ☑ нон харидан  ☐ дарс хондан  ☐ занг задан
completeTask(5);        // Чунин вазифа нест
console.log(countDone());   // 1

// санҷиш (рӯйхати нав, ҳеҷ кадом иҷро нашудааст)
completeTask(0);
completeTask(0);
console.log(countDone());   // 1
```
